--- FILE: app.py ---
from flask import Flask, jsonify, request
from flask_cors import CORS

app = Flask(__name__)
CORS(app)

# In-memory "database" of students
students = [
    {"id": 1, "name": "Alice", "grade": "A"},
    {"id": 2, "name": "Bob", "grade": "B"},
]
next_id = 3

# Health check
@app.route("/api/health")
def health():
    return jsonify({"status": "ok", "message": "Flask is running"}), 200

# Get all students
@app.route("/api/students", methods=["GET"])
def get_students():
    return jsonify(students), 200

# Get a student by ID
@app.route("/api/students/<int:sid>", methods=["GET"])
def get_student(sid):
    s = next((s for s in students if s["id"] == sid), None)
    if not s:
        return jsonify({"error": "Not found"}), 404
    return jsonify(s), 200

# Add a new student
@app.route("/api/students", methods=["POST"])
def add_student():
    global next_id
    data = request.get_json()
    if not data or "name" not in data or "grade" not in data:
        return jsonify({"error": "name and grade required"}), 400
    new_student = {"id": next_id, "name": data["name"], "grade": data["grade"]}
    students.append(new_student)
    next_id += 1
    return jsonify(new_student), 201

# Update a student
@app.route("/api/students/<int:sid>", methods=["PUT"])
def update_student(sid):
    s = next((s for s in students if s["id"] == sid), None)
    if not s:
        return jsonify({"error": "Not found"}), 404
    data = request.get_json()
    s["name"] = data.get("name", s["name"])
    s["grade"] = data.get("grade", s["grade"])
    return jsonify(s), 200

# Delete a student
@app.route("/api/students/<int:sid>", methods=["DELETE"])
def delete_student(sid):
    global students
    before = len(students)
    students = [s for s in students if s["id"] != sid]
    if len(students) == before:
        return jsonify({"error": "Not found"}), 404
    return jsonify({"message": "Deleted"}), 200

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000, debug=True)

--- FILE: test_app.py ---
import pytest
from app import app

@pytest.fixture
def client():
    app.config["TESTING"] = True
    with app.test_client() as client:
        yield client

def test_health_check(client):
    res = client.get("/api/health")
    assert res.status_code == 200
    assert res.get_json()["status"] == "ok"

def test_get_all_students(client):
    res = client.get("/api/students")
    assert res.status_code == 200
    assert isinstance(res.get_json(), list)

def test_get_student_found(client):
    res = client.get("/api/students/1")
    assert res.status_code == 200
    data = res.get_json()
    assert data["id"] == 1
    assert "name" in data
    assert "grade" in data

def test_get_student_not_found(client):
    res = client.get("/api/students/999")
    assert res.status_code == 404

def test_add_student(client):
    payload = {"name": "Ali", "grade": "A"}
    res = client.post("/api/students", json=payload)
    assert res.status_code == 201
    data = res.get_json()
    assert data["name"] == payload["name"]
    assert data["grade"] == payload["grade"]
    assert "id" in data

def test_add_student_missing_field(client):
    res = client.post("/api/students", json={"name": "John"})
    assert res.status_code == 400

def test_update_student(client):
    payload = {"name": "Charlie", "grade": "B+"}
    res = client.put("/api/students/1", json=payload)
    assert res.status_code == 200
    data = res.get_json()
    assert data["name"] == "Charlie"
    assert data["grade"] == "B+"

def test_delete_student(client):
    res = client.delete("/api/students/2")
    assert res.status_code == 200
    assert res.get_json()["message"] == "Deleted"
    # Confirm deletion
    res_check = client.get("/api/students/2")
    assert res_check.status_code == 404

--- FILE: .github/workflows/ci.yml ---
(First run in terminal: mkdir -p .github/workflows)
(Then create this file inside that folder in VS Code)

name: CI/CD Pipeline

on:
  push:
    branches: [ main ]

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout code
        uses: actions/checkout@v3

      - name: Run tests
        run: |
          pip install flask flask-cors pytest
          pytest test_app.py -v

      - name: Log in to Docker Hub
        uses: docker/login-action@v2
        with:
          username: ${{ secrets.DOCKERHUB_USERNAME }}
          password: ${{ secrets.DOCKERHUB_TOKEN }}

      - name: Build and push Docker image
        uses: docker/build-push-action@v4
        with:
          context: .
          push: true
          tags: ${{ secrets.DOCKERHUB_USERNAME }}/flask-api:latest

      - name: Deploy to EC2
        uses: appleboy/ssh-action@v1.0.0
        with:
          host: ${{ secrets.EC2_HOST }}
          username: ubuntu
          key: ${{ secrets.EC2_SSH_KEY }}
          script: |
            docker pull ${{ secrets.DOCKERHUB_USERNAME }}/flask-api:latest
            docker stop flask-api || true
            docker rm flask-api || true
            docker run -d -p 5000:5000 --restart=always \
              --name flask-api \
              ${{ secrets.DOCKERHUB_USERNAME }}/flask-api:latest
            docker ps

================================================================================
THEORY QUICK ANSWERS
================================================================================

CI/CD:
CI = auto-tests code on every push
CD = auto-deploys tested code to server, no manual steps

Docker container vs VM:
VM = full OS with own kernel, heavy, slow
Container = shares host OS kernel, lightweight, starts in seconds

Dockerfile:
Text file with instructions to build Docker image
FROM > WORKDIR > COPY > RUN > EXPOSE > CMD

GitHub Actions:
Automated workflow triggered by git push
Runs jobs on cloud VMs to test, build, deploy

REST API methods:
GET = read, POST = create, PUT = update, DELETE = delete

HTTP status codes:
200 = OK, 201 = Created, 400 = Bad Request, 404 = Not Found, 500 = Server Error

nohup:
Keeps process running after terminal closes
nohup python3 app.py > app.log 2>&1 &

--restart=always:
Docker container auto-restarts if it crashes or EC2 reboots

Security Group:
AWS virtual firewall controlling which ports are open on EC2

systemctl:
Linux service manager, starts apps on boot, restarts on crash

CMD vs RUN in Dockerfile:
RUN = executes during image BUILD (install packages)
CMD = executes when CONTAINER STARTS (run the app)

2>&1 meaning:
Redirects stderr (errors) into stdout so both go to same log file

git pull vs git fetch:
pull = download + merge automatically
fetch = download only, you merge manually
