# Build the image
docker build -t flask-api:v1 .

# Run it (detached, port mapped)
docker run -d -p 5000:5000 --name flask-api flask-api:v1

# Verify it works
curl http://localhost:5000/api/health
curl http://localhost:5000/api/students

# Stop and remove when done testing
docker stop flask-api && docker rm flask-api
